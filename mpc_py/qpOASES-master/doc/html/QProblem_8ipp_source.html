104/*
105 * s e t A
106 */
107inline returnValue QProblem::setA( Matrix *A_new )
108{
109 int_t j;
110 int_t nV = getNV( );
111 int_t nC = getNC( );
112
113 if ( nV == 0 )
114 return THROWERROR( RET_QPOBJECT_NOT_SETUP );
115
116 if ( A_new == 0 )
117 return THROWERROR( RET_INVALID_ARGUMENTS );
118
119 /* Set constraint matrix AND update member AX. */
120 if ( ( freeConstraintMatrix == BT_TRUE ) && ( A != 0 ) )
121 {
122 delete A;
123 A = 0;
124 }
125 A = A_new;
126 freeConstraintMatrix = BT_FALSE;
127
128 A->times(1, 1.0, x, nV, 0.0, Ax, nC);
129
130 A->getRowNorm(tempC);
131
132 for( j=0; j<nC; ++j )
133 {
134 Ax_u[j] = ubA[j] - Ax[j];
135 Ax_l[j] = Ax[j] - lbA[j];
136 // (ckirches) disable constraints with empty rows
137 if ( isZero( tempC[j] ) == BT_TRUE )
138 constraints.setType ( j, ST_DISABLED );
139 }
140
141
142 return SUCCESSFUL_RETURN;
143}

149inline returnValue QProblem::setA( const real_t* const A_new )
150{
151 int_t j;
152 int_t nV = getNV( );
153 int_t nC = getNC( );
154 DenseMatrix* dA;
155
156 if ( nV == 0 )
157 return THROWERROR( RET_QPOBJECT_NOT_SETUP );
158
159 if ( A_new == 0 )
160 return THROWERROR( RET_INVALID_ARGUMENTS );
161
162 /* Set constraint matrix AND update member AX. */
163 if ( ( freeConstraintMatrix == BT_TRUE ) && ( A != 0 ) )
164 {
165 delete A;
166 A = 0;
167 }
168 A = dA = new DenseMatrix(nC, nV, nV, (real_t*) A_new);
169 freeConstraintMatrix = BT_TRUE;
170
171 A->times(1, 1.0, x, nV, 0.0, Ax, nC);
172
173 for( j=0; j<nC; ++j )
174 {
175 Ax_u[j] = ubA[j] - Ax[j];
176 Ax_l[j] = Ax[j] - lbA[j];
177 }
178
179 return SUCCESSFUL_RETURN;
180}

186inline returnValue QProblem::setLBA( const real_t* const lbA_new )
187{
188 uint_t i;
189 uint_t nV = (uint_t)getNV( );
190 uint_t nC = (uint_t)getNC( );
191
192 if ( nV == 0 )
193 return THROWERROR( RET_QPOBJECT_NOT_SETUP );
194
195 if ( lbA_new != 0 )
196 {
197 memcpy( lbA,lbA_new,nC*sizeof(real_t) );
198 }
199 else
200 {
201 /* if no lower constraints' bounds are specified, set them to -infinity */
202 for( i=0; i<nC; ++i )
203 lbA[i] = -INFTY;
204 }
205
206 return SUCCESSFUL_RETURN;
207}
