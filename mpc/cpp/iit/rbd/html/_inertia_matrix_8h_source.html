204TPL
205inline void CLASS::changeMass(Scalar newMass) {
206 // Note the use of indices AX and hard-coded 0, to make it independent from
207 // the convention angular/linear
208 this->template block<3,6>(AX,0) *= newMass/getMass();
209 block33(LX,AX) = block33(AX,LX).transpose();
210 data(LX,LX) = data(LY,LY) = data(LZ,LZ) = newMass;
211}
212
213TPL
214inline void CLASS::changeCOM(const Vec3& newcom)
215{
216 // Update the angular-linear block according to the new COM position
217 setSkewSymmetricBlock( getMass() * newcom, block33(AX,LX));
218
219 // Correct the 3x3 tensor. Use the block(AX,LX) which reflects the new COM
220 // and the block(LX,AX) which instead is still based on the previous value.
221 // I = I - m(cx)(cx)^T + m(c'x)(c'x)^T
222 // where cx is the skew symmetric matrix for the old COM vector, while
223 // c'x is the one for the new COM vector.
224 block33(AX,AX) += (
225 ( block33(AX,LX) * block33(AX,LX).transpose() ) -
226 ( block33(LX,AX).transpose() * block33(LX,AX) )
227 ) / getMass();
228 // Update the linear-angular block
229 block33(LX,AX) = block33(AX,LX).transpose();
230 // alternatively:
231 // setSkewSymmetricBlock( - getMass() * newcom, block33(LX,AX));
232}

240TPL
241inline void CLASS::set(
242 Scalar mass,
243 const Vec3& com,
244 const Mat33& tensor)
245{
246 block33(AX,AX) = tensor;// + mass * comCrossMx * comCrossMx.transpose();
247
248 data(AX,LY) = data(LY,AX) = - ( data(AY,LX) = data(LX,AY) = mass*com(Z) );
249 data(AZ,LX) = data(LX,AZ) = - ( data(AX,LZ) = data(LZ,AX) = mass*com(Y) );
250 data(AY,LZ) = data(LZ,AY) = - ( data(AZ,LY) = data(LY,AZ) = mass*com(X) );
251 // Equivalent, just slightly less efficient (probably because of Eigen's blocks and repeated product):
252 //setSkewSymmetricBlock( mass * com, block33(AX,LX));
253 //setSkewSymmetricBlock(- mass * com, block33(LX,AX));
254
255 data(LX,LX) = data(LY,LY) = data(LZ,LZ) = mass;
256}

258TPL
259template<typename OtherDerived>
260inline CLASS& CLASS::operator=
261 (const MatrixBase<OtherDerived>& other)
262{
263 // Here we silently assume that 'other' is also an inertia...
264 // Type safety would suggest to prevent the assignment of anything but
265 // another inertia, but that would prevent, for example, the assignment
266 // of matrix expressions. We also do not want to perform any check, for
267 // performance reasons (remember this library is meant primarily to support
268 // code generation, not to be a robust API for user applications).
269 block33(AX,AX) = other.template block<3,3>(AX,AX);
270 data(AX,LY) = data(LY,AX) = - ( data(AY,LX) = data(LX,AY) = other(LX,AY) );
271 data(AZ,LX) = data(LX,AZ) = - ( data(AX,LZ) = data(LZ,AX) = other(LZ,AX) );
272 data(AY,LZ) = data(LZ,AY) = - ( data(AZ,LY) = data(LY,AZ) = other(LY,AZ) );
273 data(LX,LX) = data(LY,LY) = data(LZ,LZ) = other(LX,LX);
274 return *this;
275}

277TPL
278template<typename OtherDerived>
279inline CLASS& CLASS::operator+=
280 (const MatrixBase<OtherDerived>& other)
281{
282 block33(AX,AX) += other.template block<3,3>(AX,AX);
283 data(AX,LY) = data(LY,AX) = - ( data(AY,LX) = (data(LX,AY) += other(LX,AY)) );
284 data(AZ,LX) = data(LX,AZ) = - ( data(AX,LZ) = (data(LZ,AX) += other(LZ,AX)) );
285 data(AY,LZ) = data(LZ,AY) = - ( data(AZ,LY) = (data(LY,AZ) += other(LY,AZ)) );
286 data(LX,LX) = data(LY,LY) = (data(LZ,LZ) += other(LX,LX));
287 return *this;
288}

290TPL
291inline void CLASS::setTheFixedZeros()
292{
293 block33(LX,LX).setZero(); // the diagonal won't be zero, but who cares
294 data(AX,LX) = data(AY,LY) = data(AZ,LZ) =
295 data(LX,AX) = data(LY,AY) = data(LZ,AZ) = 0;
296}

298TPL
299template<typename Vector>
300inline void CLASS::setSkewSymmetricBlock(
301 const MatrixBase<Vector>& v, Block33_t block)
302{
303 block(X,Y) = - ( block(Y,X) = v(Z) );
304 block(Z,X) = - ( block(X,Z) = v(Y) );
305 block(Y,Z) = - ( block(Z,Y) = v(X) );
306}
