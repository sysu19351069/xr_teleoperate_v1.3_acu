69 template <typename Derived>
70 static void fillAsRotationMatrix(double rx, double ry, double rz,
71 const MatrixBase<Derived>& mx)
72 {
73 eigen_assert(mx.rows() == 3 && mx.cols() == 3);
74
75 double cx = std::cos(rx);
76 double sx = std::sin(rx);
77 double cy = std::cos(ry);
78 double sy = std::sin(ry);
79 double cz = std::cos(rz);
80 double sz = std::sin(rz);
81
82 const_cast<MatrixBase<Derived>&>(mx)
83 << cy*cz, cx*sz+sx*sy*cz, sx*sz-cx*sy*cz,
84 -cy*sz, cx*cz-sx*sy*sz, cx*sy*sz+sx*cz,
85 sy, -sx*cy, cx*cy;
86 }

92 static Matrix33d buildRotationMatrix(double rx, double ry, double rz) {
93 double cx = std::cos(rx);
94 double sx = std::sin(rx);
95 double cy = std::cos(ry);
96 double sy = std::sin(ry);
97 double cz = std::cos(rz);
98 double sz = std::sin(rz);
99 Matrix33d ret;
100 ret << cy*cz, cx*sz+sx*sy*cz, sx*sz-cx*sy*cz,
101 -cy*sz, cx*cz-sx*sy*sz, cx*sy*sz+sx*cz,
102 sy, -sx*cy, cx*cy;
103 return ret;
104 }

167 template <typename Derived, typename Other>
168 static typename Core<typename Other::Scalar>::Vector3 transform(
169 const MatrixBase<Derived>& homT,
170 const MatrixBase<Other>& vect3d)
171 {
172 eigen_assert(homT.rows() == 4 && homT.cols() == 4); // weak way to check if it is a homogeneous transform
173 eigen_assert(vect3d.rows() == 3 && vect3d.cols() == 1); // check if it is a 3d column
174 return rotationMx(homT) * vect3d + positionVector(homT);
175 }
